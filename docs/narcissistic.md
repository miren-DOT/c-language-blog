\# C语言：水仙花数

\## 题目说明

水仙花数：一个三位数，每一位数字的立方相加等于数字本身。

例如：153 = 1³ + 5³ + 3³



\## C语言代码（只用循环，无自定义函数）

```c

\#include <stdio.h>

int main()

{

&#x20;   int num, a, b, c;

&#x20;   for(num = 100; num <= 999; num++)

&#x20;   {

&#x20;       a = num / 100;        // 百位

&#x20;       b = num / 10 % 10;    // 十位

&#x20;       c = num % 10;         // 个位

&#x20;       if(a\*a\*a + b\*b\*b + c\*c\*c == num)

&#x20;       {

&#x20;           printf("%d 是水仙花数\\n",num);

&#x20;       }

&#x20;   }

&#x20;   return 0;

}
