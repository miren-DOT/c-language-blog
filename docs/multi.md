\# 九九乘法表（C语言循环示例）

\## 题目说明

使用双层for循环，打印九九乘法表。仅使用循环语句，适合初学者练习嵌套循环。



```c

\#include <stdio.h>

int main()

{

&#x20;   int i,j;

&#x20;   // i代表行

&#x20;   for(i = 1; i <= 9; i++)

&#x20;   {

&#x20;       // j代表列，j不超过i，实现上三角效果

&#x20;       for(j = 1; j <= i; j++)

&#x20;       {

&#x20;           printf("%d\*%d=%d\\t",j,i,i\*j);

&#x20;       }

&#x20;       // 一行结束，换行

&#x20;       printf("\\n");

&#x20;   }

&#x20;   return 0;

}
